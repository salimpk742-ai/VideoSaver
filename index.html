async function downloadFile(item, button) { if (!item || !item.url) { showMessage("This quality has no downloadable file.", "error"); return; } button.disabled = true; button.textContent = "Opening..."; const link = document.createElement("a"); link.href = item.url; link.target = "_blank"; link.rel = "noopener"; link.download = makeFilename(item); document.body.appendChild(link); link.click(); link.remove(); showMessage("Download opened. If it does not start, use Save in the new tab.", "success"); button.disabled = false; button.textContent = "Download"; }
